require("dotenv").config();
const express = require("express");
const multer = require("multer");
const OpenAI = require("openai");
const pdfPoppler = require("pdf-poppler");
const fs = require("fs");
const path = require("path");
const verifyLogin = require("../middlewares/verifyLogin.js");
// Use Capitalized name for Mongoose models by convention
const PastPaper = require("../models/PastPaper.js");
const { getAuth } = require('firebase-admin/auth');
const User = require('../models/User');

const router = express.Router();

// 1. Ensure the root "uploads" directory exists
const uploadDir = path.join(process.cwd(), "uploads");
if (!fs.existsSync(uploadDir)) {
  fs.mkdirSync(uploadDir, { recursive: true });
}

// 2. Configure Disk Storage with unique filenames to prevent overwriting
const storage = multer.diskStorage({
  destination: (req, file, cb) => {
    cb(null, uploadDir);
  },
  filename: (req, file, cb) => {
    const ext = path.extname(file.originalname);
    const baseName = path.basename(file.originalname, ext).replace(/\s+/g, "_");
    const uniqueSuffix = `${Date.now()}-${Math.round(Math.random() * 1e9)}`;
    cb(null, `${baseName}-${uniqueSuffix}${ext}`);
  },
});

// 3. Initialize Multer with the custom storage
const upload = multer({
  storage: storage,
  limits: { fileSize: 5 * 1024 * 1024 }, // 5 MB limit
  fileFilter: (req, file, cb) => {
    if (file.mimetype === 'application/pdf') {
      cb(null, true);
    } else {
      cb(new Error('Only PDF files are allowed!'), false);
    }
  }
});

// Initialize OpenAI client
const openai = new OpenAI({
  apiKey: process.env.OPENAI_API_KEY,
});

router.post("/process", upload.single("file"), async (req, res) => {

  // Extract metadata sent from the frontend FormData (Added 'date' here)
  const { subject, date } = req.body;
  // 1. Extract token from the Authorization header
  const authHeader = req.headers.authorization;

  if (!authHeader || !authHeader.startsWith("Bearer ")) {
    console.log("Token not provided or invalid format");
    return res.status(401).json({ error: "Unauthorized: No token provided" });
  }

  // Split "Bearer <token>" to get just the token string
  const token = authHeader.split(" ")[1];

  // 2. Verify the token using Firebase Admin
  const decodedToken = await getAuth().verifyIdToken(token);

  // Extract data from the Firebase token
  const { uid, email, name, picture } = decodedToken;

  console.log(uid);
  const user = await User.findOne({ uid });
  if (!user) {
    return res.status(401).json({ error: "user not found in database" });
  }

  if (!req.file) {
    return res.status(400).json({ success: false, error: "No PDF file uploaded" });
  }

  // 1. Check if the subject already exists (case-insensitive check)
  const subExist = user.workspaces.some(
    (ws) => ws.trim().toLowerCase() === subject.trim().toLowerCase()
  );

  if (subExist) {
    return res.status(400).json({
      success: "false",
      error: `Workspace '${subject}' already exists`
    });
  };


  const pdfPath = req.file.path;
  const originalName = req.file.originalname;
  const savedFilename = req.file.filename; // The unique filename generated by multer

  // Create a unique temporary directory for the images of this specific PDF
  const outputDir = path.join(uploadDir, `extracted_${Date.now()}`);

  try {
    // Ensure extraction folder exists
    if (!fs.existsSync(outputDir)) {
      fs.mkdirSync(outputDir, { recursive: true });
    }

    // 1. Convert all pages of PDF to individual JPEG images
    const popplerOptions = {
      format: "jpeg",
      out_dir: outputDir,
      out_prefix: "page",
      page: null,
    };
    await pdfPoppler.convert(pdfPath, popplerOptions);

    // 2. Get all extracted page images sorted numerically
    const imageFiles = fs
      .readdirSync(outputDir)
      .filter((file) => file.endsWith(".jpg") || file.endsWith(".jpeg"))
      .sort((a, b) => a.localeCompare(b, undefined, { numeric: true }));

    const savedDocs = [];

    // 3. Process each page sequentially (Creates one document PER PAGE)
    for (let i = 0; i < imageFiles.length; i++) {
      const imagePath = path.join(outputDir, imageFiles[i]);
      const imageBuffer = fs.readFileSync(imagePath);
      const base64Image = imageBuffer.toString("base64");

      // Step A: Extract text & equations with gpt-4o-mini
      const visionResponse = await openai.chat.completions.create({
        model: "gpt-4o-mini",
        messages: [
          {
            role: "system",
            content:
              "Extract all questions, text, choices, and mathematical formulas from this exam page. Maintain clean numbering and formatting.",
          },
          {
            role: "user",
            content: [
              { type: "text", text: "Extract all content from this exam paper image." },
              {
                type: "image_url",
                image_url: {
                  url: `data:image/jpeg;base64,${base64Image}`,
                  detail: "high",
                },
              },
            ],
          },
        ],
      });

      const extractedText = visionResponse.choices[0].message.content;

      // Step B: Generate 1536-dimension embeddings
      const embeddingResponse = await openai.embeddings.create({
        model: "text-embedding-3-small",
        input: extractedText,
        dimensions: 1536,
      });

      const vector = embeddingResponse.data[0].embedding;

      // Step C: Insert into MongoDB using your Mongoose Schema
      const pageDoc = new PastPaper({
        subject: subject || "Unknown Subject",
        date: date || "Unknown Date", // This will no longer throw a ReferenceError
        fileName: originalName,
        pageNumber: i + 1,
        original_text: extractedText,
        embeddings: vector,
      });

      await pageDoc.save();
      savedDocs.push({ page: i + 1, length: extractedText.length });
    }

    user.workspaces.push(subject.trim());
    await user.save();

    res.status(200).json({
      success: true,
      message: `workspace ${subject} created!`,
      pages: savedDocs,
    });
  } catch (error) {
    console.error("PDF Processing Error:", error);
    res.status(500).json({ success: "false", error: error.message });
  } finally {
    // Cleanup temporary files and directory (Deletes the PDF and the JPEGs to save space)
    if (fs.existsSync(pdfPath)) fs.unlinkSync(pdfPath);
    if (fs.existsSync(outputDir)) fs.rmSync(outputDir, { recursive: true, force: true });
  }
});
